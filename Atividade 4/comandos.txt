       (carol:Pessoa {nome: "Carol", pais: "Brasil"}),
       (foto:Foto {legenda: "Meu cachorro"}),
       (gabriel)-[:SEGUE]->(carol),
       (carol)-[:SEGUE]->(gabriel),
       (carol)-[:ADICIONOU]->(foto),
       (gabriel)-[:CURTIU {data: "2019-01-01"}]->(foto)
RETURN gabriel, carol, foto
